file(GLOB_RECURSE LMNS_SRC CONFIGURE_DEPENDS *.cpp)
file(GLOB_RECURSE LMNS_HEAD CONFIGURE_DEPENDS *.hpp)

add_library(lmns STATIC ${LMNS_SRC} ${LMNS_HEAD})

target_include_directories(lmns
	PUBLIC
		$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
		$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
	PRIVATE
		${CMAKE_CURRENT_SOURCE_DIR}/src
)

target_link_libraries(lmns PUBLIC lmns_options)

target_compile_options(lmns PRIVATE
	$<$<CXX_COMPILER_ID:GNU>:
		-Werror
	>
)

install(TARGETS lmns
	EXPORT lmns_targets
	ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
	LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
	RUNTIME DESTINATION #{CMAKE_INSTALL_BINDIR}
)

install(FILES ${LMNS_HEAD} DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/lmns)
